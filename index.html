export default { async fetch(request, env) { const url = new URL(request.url); const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, PUT, DELETE, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" }; if (request.method === "OPTIONS") return new Response("OK", { headers: corsHeaders }); // CREATE (POST) if (request.method === "POST") { const formData = await request.formData(); const matricula = formData.get("matricula"); const registro = { matricula, nombre: "Admin", fecha: new Date().toLocaleString() }; await env.CONTACTOS.put(matricula, JSON.stringify(registro)); return new Response(`✅ Guardado`, { headers: corsHeaders }); } // READ (GET) if (request.method === "GET") { const list = await env.CONTACTOS.list(); let records = []; for (let key of list.keys) { const value = await env.CONTACTOS.get(key.name); records.push(JSON.parse(value)); } return new Response(JSON.stringify(records), { headers: { ...corsHeaders, "Content-Type": "application/json" } }); } // UPDATE (PUT) if (request.method === "PUT") { const body = await request.json(); if (body.matricula && body.nombre) { const registro = { matricula: body.matricula, nombre: body.nombre, fecha: new Date().toLocaleString() }; await env.CONTACTOS.put(body.matricula, JSON.stringify(registro)); return new Response(`✅ Actualizado`, { headers: corsHeaders }); } return new Response("Faltan datos", { status: 400, headers: corsHeaders }); } // DELETE (DELETE) if (request.method === "DELETE") { const matriculaABorrar = url.searchParams.get("matricula"); if (matriculaABorrar) { await env.CONTACTOS.delete(matriculaABorrar); return new Response(`🗑️ Registro eliminado.`, { headers: corsHeaders }); } } return new Response("API CRUD Activa", { headers: corsHeaders }); } };
